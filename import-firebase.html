    Copy the Firebase rules

    Select chaman-lal-dbms → Firestore Database → (default) → Rules. Paste the complete code below and Publish. These are Firestore rules.

    rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function owner(uid) {
      return request.auth != null && request.auth.uid == uid;
    }
    function migrated(uid) {
      return exists(/databases/$(database)/documents/users/$(uid)/settings/catalog);
    }
    function commonRecord(d) {
      return d.schemaVersion == 2 && d.deleted is bool
        && d.order is int && d.order >= 0
        && d.rev is int && d.rev > 0
        && d.updatedAt == request.time;
    }

    match /users/{uid}/cities/{cityId} {
      function validCity() {
        let d = request.resource.data;
        return d.keys().hasAll(['name', 'order', 'deleted', 'rev', 'schemaVersion', 'createdAt', 'updatedAt'])
          && d.keys().hasOnly(['name', 'order', 'deleted', 'rev', 'schemaVersion', 'createdAt', 'updatedAt'])
          && commonRecord(d)
          && d.name is string && d.name.size() > 0 && d.name.size() <= 80;
      }
      allow read: if owner(uid);
      allow create: if owner(uid) && validCity()
        && request.resource.data.rev == 1 && request.resource.data.createdAt == request.time;
      allow update: if owner(uid) && migrated(uid) && validCity()
        && request.resource.data.rev == resource.data.rev + 1
        && request.resource.data.createdAt == resource.data.createdAt;
      // Delete in the app sets deleted=true. Restore sets it back to false.
      allow delete: if false;
    }

    match /users/{uid}/doctors/{doctorId} {
      function validDoctor() {
        let d = request.resource.data;
        return d.keys().hasAll(['name', 'cityId', 'order', 'details', 'fieldOrder', 'deleted', 'rev', 'schemaVersion', 'createdAt', 'updatedAt'])
          && d.keys().hasOnly(['name', 'cityId', 'order', 'details', 'fieldOrder', 'deleted', 'rev', 'schemaVersion', 'createdAt', 'updatedAt'])
          && commonRecord(d) && doctorId.matches('^D[0-9]+$')
          && d.name is string && d.name.size() > 0 && d.name.size() <= 200
          && d.cityId is string && d.cityId.size() > 0 && d.cityId.size() <= 1500
          && d.details is map && d.details.size() >= 2 && d.details.size() <= 50
          && d.details['Doctor Name'] == d.name
          && d.details['City'] is string && d.details['City'].size() > 0
          && d.fieldOrder is list && d.fieldOrder.size() == d.details.size()
          && d.details.keys().hasAll(d.fieldOrder) && d.details.keys().hasOnly(d.fieldOrder);
      }
      allow read: if owner(uid);
      allow create: if owner(uid) && validDoctor()
        && request.resource.data.rev == 1 && request.resource.data.createdAt == request.time;
      allow update: if owner(uid) && migrated(uid) && validDoctor()
        && request.resource.data.rev == resource.data.rev + 1
        && request.resource.data.createdAt == resource.data.createdAt;
      allow delete: if false;
    }

    match /users/{uid}/settings/catalog {
      function validCatalog() {
        let d = request.resource.data;
        return d.keys().hasAll(['schemaVersion', 'status', 'headers', 'sourceFile', 'sourceSheet', 'sourceHash', 'sourceRowCount', 'sourceColumnCount', 'importedCityCount', 'importedDoctorCount', 'legacyRevision', 'importedAt'])
          && d.keys().hasOnly(['schemaVersion', 'status', 'headers', 'sourceFile', 'sourceSheet', 'sourceHash', 'sourceRowCount', 'sourceColumnCount', 'importedCityCount', 'importedDoctorCount', 'legacyRevision', 'importedAt'])
          && d.schemaVersion == 2 && d.status == 'ready'
          && d.headers is list && d.headers.size() == 16
          && d.sourceFile == 'Database.xlsx' && d.sourceSheet == 'Doctors'
          && d.sourceHash is string && d.sourceHash.matches('^[a-f0-9]{64}$')
          && d.sourceRowCount == 195 && d.sourceColumnCount == 16
          && d.importedCityCount is int && d.importedCityCount >= 6
          && d.importedDoctorCount is int && d.importedDoctorCount >= 195
          && d.legacyRevision is int && d.legacyRevision >= 0
          && d.importedAt == request.time;
      }
      allow read: if owner(uid);
      allow create: if owner(uid) && validCatalog();
      // Prevent a later import from resetting the cloud directory.
      allow update, delete: if false;
    }

    // Preserve existing visit history and the same stable doctor IDs.
    match /users/{uid}/visits/{visitId} {
      allow read: if owner(uid);
      allow create: if owner(uid)
        && request.resource.data.keys().hasAll(['doctorId', 'doctorName', 'city', 'visitDate', 'createdAt', 'schemaVersion'])
        && request.resource.data.keys().hasOnly(['doctorId', 'doctorName', 'city', 'visitDate', 'createdAt', 'schemaVersion'])
        && request.resource.data.doctorId is string && request.resource.data.doctorId.matches('^D[0-9]+$')
        && request.resource.data.doctorName is string && request.resource.data.doctorName.size() > 0 && request.resource.data.doctorName.size() <= 200
        && request.resource.data.city is string && request.resource.data.city.size() > 0 && request.resource.data.city.size() <= 160
        && request.resource.data.visitDate is string && request.resource.data.visitDate.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
        && visitId == request.resource.data.doctorId + '__' + request.resource.data.visitDate
        && request.resource.data.createdAt == request.time && request.resource.data.schemaVersion == 1;
      allow update: if false;
      allow delete: if owner(uid);
    }

    // Compatibility with the current HTML until the one-time import finishes.
    // Once catalog exists, the old overlay becomes read-only to avoid two
    // directories accepting different edits. Existing data is kept intact.
    match /users/{uid}/settings/directory {
      function validLegacy() {
        let d = request.resource.data;
        return d.keys().hasAll(['cities', 'doctors', 'revision', 'schemaVersion', 'updatedAt'])
          && d.keys().hasOnly(['cities', 'doctors', 'revision', 'schemaVersion', 'updatedAt'])
          && d.cities is map && d.doctors is map
          && d.revision is int && d.revision > 0
          && d.schemaVersion == 1 && d.updatedAt == request.time;
      }
      allow read: if owner(uid);
      allow create: if owner(uid) && !migrated(uid) && validLegacy()
        && request.resource.data.revision == 1;
      allow update: if owner(uid) && !migrated(uid) && validLegacy()
        && request.resource.data.revision == resource.data.revision + 1;
      allow delete: if false;
    }
  }
}
